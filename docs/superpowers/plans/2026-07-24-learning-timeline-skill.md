# Learning Timeline Skill Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a repository-local Codex skill that discovers newly added solution packages, writes one evidence-based learning entry per package into today's timeline, and exposes the timeline from the generated solution index.

**Architecture:** A shared Python script performs deterministic, read-only Git discovery and emits JSON. The Codex skill and Cursor command read that JSON plus package source files, write the judgment-heavy summaries and timeline state, and verify idempotency. `tools/indexgen` only renders a stable link to the independently maintained timeline index.

**Tech Stack:** Codex skill Markdown and `agents/openai.yaml`, Python 3 standard library and `unittest`, Go standard library and `testing`, Git.

#!/usr/bin/env python3

import json
import os
from pathlib import Path
import subprocess
import sys
import tempfile
import unittest


SCRIPT = Path(__file__).with_name("discover.py")


def git(repo, *args):
    return subprocess.run(
        ["git", *args],
        cwd=repo,
        check=True,
        text=True,
        capture_output=True,
    ).stdout.strip()


def write(repo, relative, content):
    path = Path(repo, relative)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(content)


def commit(repo, message):
    git(repo, "add", ".")
    env = os.environ | {
        "GIT_AUTHOR_NAME": "Timeline Test",
        "GIT_AUTHOR_EMAIL": "timeline@example.com",
        "GIT_COMMITTER_NAME": "Timeline Test",
        "GIT_COMMITTER_EMAIL": "timeline@example.com",
    }
    subprocess.run(
        ["git", "commit", "-q", "-m", message],
        cwd=repo,
        check=True,
        env=env,
    )
    return git(repo, "rev-parse", "HEAD")


def add_package(repo, package):
    write(repo, f"{package}/solution.go", "package main\n")
    write(repo, f"{package}/solution_test.go", "package main\n")
    write(repo, f"{package}/problem.md", "# Problem\n")


class DiscoverTest(unittest.TestCase):
    def setUp(self):
        self.temp = tempfile.TemporaryDirectory()
        self.repo = self.temp.name
        git(self.repo, "init", "-q")
        write(self.repo, ".gitignore", "ignored/\n")
        commit(self.repo, "base")

    def tearDown(self):
        self.temp.cleanup()

    def discover(self, expect_ok=True):
        result = subprocess.run(
            [sys.executable, str(SCRIPT), "--repo", self.repo],
            text=True,
            capture_output=True,
        )
        if expect_ok and result.returncode != 0:
            self.fail(result.stderr)
        return result

Create `tools/timeline/discover.py`:

```python
#!/usr/bin/env python3

import argparse
import json
from pathlib import Path
import re
import subprocess
import sys


STATE_RE = re.compile(r"<!--\s*through-commit:\s*([0-9a-fA-F]{40})\s*-->")
LINK_RE = re.compile(r"\]\(\.\./\.\./(src/[^)\s]+?)/?\)")
SOURCE_NAMES = (
    "solution.go",
    "solution_test.go",
    "problem.md",
    "readme.md",
    "README.md",
)


class DiscoveryError(Exception):
    pass


def run_git(repo, *args, check=True):
    result = subprocess.run(
        ["git", *args],
        cwd=repo,
        text=True,
        capture_output=True,
    )
    if check and result.returncode != 0:
        detail = result.stderr.strip() or result.stdout.strip()
        raise DiscoveryError(f"git {' '.join(args)}: {detail}")
    return result


def timeline_dir(repo):
    return Path(repo, "docs", "timeline")


def read_through_commit(repo):
    path = timeline_dir(repo) / "README.md"
    if not path.exists():
        return ""
    match = STATE_RE.search(path.read_text())
    return match.group(1).lower() if match else ""


def latest_solution_baseline(repo):
    result = run_git(
        repo,
        "log",
        "-1",
        "--format=%H",
        "--diff-filter=A",
        "--",
        ":(glob)src/**/solution.go",
    )
    solution_commit = result.stdout.strip()
    if not solution_commit:
        raise DiscoveryError("no committed solution.go addition found")
    parent = run_git(repo, "rev-parse", f"{solution_commit}^")
    return parent.stdout.strip()


def validate_commit(repo, commit):
    result = run_git(repo, "cat-file", "-e", f"{commit}^{{commit}}", check=False)
    if result.returncode != 0:
        raise DiscoveryError(f"invalid through-commit: {commit}")


def added_files(repo, baseline):
    committed = set(
        filter(
            None,
            run_git(
                repo,
                "diff",
                "--name-only",
                "--diff-filter=A",
                f"{baseline}..HEAD",
                "--",
            ).stdout.splitlines(),
        )
    )
    staged = set(
        filter(
            None,
            run_git(
                repo,
                "diff",
                "--cached",
                "--name-only",
                "--diff-filter=A",
                "--",
            ).stdout.splitlines(),
        )
    )
    untracked = set(
        filter(
            None,
            run_git(
                repo,
                "ls-files",
                "--others",
                "--exclude-standard",
                "--",
            ).stdout.splitlines(),
        )
    )
    return {
        "committed": committed,
        "staged": staged,
        "untracked": untracked,
    }


def recorded_packages(repo):
    found = set()
    directory = timeline_dir(repo)
    if not directory.exists():
        return found
    for path in sorted(directory.glob("[0-9][0-9][0-9][0-9]-*.md")):
        for package in LINK_RE.findall(path.read_text()):
            found.add(package.rstrip("/"))
    return found


def discover(repo):
    repo = Path(repo).resolve()
    head = run_git(repo, "rev-parse", "HEAD").stdout.strip()
    baseline = read_through_commit(repo)
    if baseline:
        validate_commit(repo, baseline)
    else:
        baseline = latest_solution_baseline(repo)

    origins = added_files(repo, baseline)
    already_recorded = recorded_packages(repo)
    packages = {}
    for origin, files in origins.items():
        for path in files:
            if not path.startswith("src/") or not path.endswith("/solution.go"):
                continue
            package = str(Path(path).parent).replace("\\", "/")
            if package in already_recorded:
                continue
            packages.setdefault(package, set()).add(origin)

    items = []
    for package in sorted(packages):
        files = [
            f"{package}/{name}"
            for name in SOURCE_NAMES
            if Path(repo, package, name).is_file()
        ]
        items.append(
            {
                "path": package,
                "origins": [
                    origin
                    for origin in ("committed", "staged", "untracked")
                    if origin in packages[package]
                ],
                "files": files,
            }
        )

    return {
        "baseline": baseline,
        "head": head,
        "packages": items,
    }


def main():
    parser = argparse.ArgumentParser(
        description="Discover newly added learn-go solution packages."
    )
    parser.add_argument("--repo", default=".", help="repository root")
    args = parser.parse_args()
    try:
        result = discover(args.repo)
    except DiscoveryError as error:
        print(f"discover timeline: {error}", file=sys.stderr)
        return 2
    json.dump(result, sys.stdout, indent=2)
    sys.stdout.write("\n")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

# Build Learning Timeline

## Core rule

Use deterministic discovery for package selection and source-grounded judgment
for summaries. Never alter solution packages while producing the timeline.

## Workflow

1. Resolve the target repository root as `REPO_ROOT`. Run:

   ```bash
   python3 "$REPO_ROOT/tools/timeline/discover.py" --repo "$REPO_ROOT"
   ```

2. Treat its JSON package list as authoritative. Do not substitute `git status`,
   file modification times, or an independently chosen commit range.

3. For every package, read all paths in `files`. Follow only directly referenced
   package artifacts when they materially explain the solution.

4. Draft one compact entry per package:

   ```markdown
   - [Problem title](../../src/platform/package/)
     — One short, evidence-based learning summary.
   ```

   Prefer the most reusable insight: algorithm, proof step, complexity
   breakthrough, implementation spike, rejected flaw, or pitfall. These are
   optional dimensions, not required labels. Do not invent criticism or repeat
   the problem statement.

5. Use the local date for `docs/timeline/YYYY-MM-DD.md`. Create the heading
   `# Learning Timeline — YYYY-MM-DD` when absent. Preserve existing content and
   append only missing packages.

6. Update `docs/timeline/README.md` only after the dated file is complete:

   ```markdown
   # Learning Timeline

   <!-- through-commit: FULL_HEAD_FROM_DISCOVERY -->

   - [YYYY-MM-DD](YYYY-MM-DD.md)
   ```

   Preserve other date links, sort them descending, and replace the marker with
   the exact `head` value from discovery. If no packages were found, create no
   dated file but still advance a valid marker.

7. Run discovery again. It must return an empty `packages` array. If it does not,
   stop and fix timeline links or state rather than duplicating entries.

## Failure rules

- On Git or invalid-state errors, write nothing.
- If evidence is insufficient for a useful package summary, report the package
  and leave it unrecorded; do not advance state past unresolved packages.
- Never rewrite earlier dated files.
- Never run `tools/indexgen` as part of a timeline update.

## Final report

Report the dated file, packages recorded, skipped packages with reasons, and the
verification result.
```
